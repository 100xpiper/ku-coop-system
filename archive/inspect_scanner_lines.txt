8254:         // Haversine Distance Calculation Formula (Returns distance in meters)
8255:         function calculateHaversineDistance(lat1, lon1, lat2, lon2) {
8256:             const R = 6371e3; // Earth radius in meters
8257:             const phi1 = (lat1 * Math.PI) / 180;
8258:             const phi2 = (lat2 * Math.PI) / 180;
8259:             const deltaPhi = ((lat2 - lat1) * Math.PI) / 180;
8260:             const deltaLambda = ((lon2 - lon1) * Math.PI) / 180;
8261: 
8262:             const a = Math.sin(deltaPhi / 2) * Math.sin(deltaPhi / 2) +
8263:                       Math.cos(phi1) * Math.cos(phi2) *
8264:                       Math.sin(deltaLambda / 2) * Math.sin(deltaLambda / 2);
8265:             const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
8266: 
8267:             return Math.round(R * c); // Distance in meters
8268:         }
